using Komet.Api;
using Vintagestory.API.Client;
using Vintagestory.API.Common;

namespace MyShaders;

// Every call into Komet sits in this class, through KAPI 1.0. A connection lasts one world: Komet takes the
// registrations back when the world closes, so StartClientSide connects again on every join.
internal static class KometBridge
{
    public static void Start(ModSystem system, ICoreClientAPI api)
    {
        // The mod id comes from modinfo.json ("myshaders"), the same as the komet.json pack's domain: Komet hands a
        // mod's defines and uniforms to its own passes
        var komet = Kapi.Connect(system, "1.0");
        if (komet is null) return; // a Komet without KAPI 1.0 (the log says why): komet.json still works on its own

        // A define for one pass only ("pass:<id>"): haze.fsh takes its day/night branch
        komet.Shaders.Define("MYSHADERS_CODE", () => "1", "pass:haze");

        // A live uniform, asked once per frame; every program and pass that declares it gets it
        komet.Shaders.Uniform("myshadersHour", () => api.World.Calendar.HourOfDay);

        // Komet rebuilds the shaders itself when the renderer switches; this is only for the log
        komet.Shaders.RendererChanged += (_, _) =>
            api.Logger.Notification("myshaders: now drawing with {0}", KapiShaders.Renderer);
    }
}
